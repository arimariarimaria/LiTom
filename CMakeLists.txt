# Keep left display rendering out of the key-send path and batch I1 bitmap writes.
if(CONFIG_ZMK_SPLIT AND NOT CONFIG_ZMK_SPLIT_ROLE_CENTRAL AND CONFIG_ZMK_DISPLAY)
  set_property(GLOBAL PROPERTY LITOM_MODULE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
  function(litom_patch_peripheral_oled)
    get_property(litom_dir GLOBAL PROPERTY LITOM_MODULE_DIR)
    get_property(libraries GLOBAL PROPERTY ZEPHYR_LIBS)
    set(replaced 0)
    foreach(library IN LISTS libraries)
      get_target_property(sources ${library} SOURCES)
      if(NOT sources)
        continue()
      endif()
      get_target_property(source_dir ${library} SOURCE_DIR)
      set(updated_sources)
      foreach(source IN LISTS sources)
        get_filename_component(absolute "${source}" ABSOLUTE BASE_DIR "${source_dir}")
        if(absolute MATCHES "/tom_oled/widgets/peripheral_status\\.c$")
          set(generated "${CMAKE_BINARY_DIR}/litom/peripheral_status.c")
          execute_process(COMMAND ${PYTHON_EXECUTABLE}
            "${litom_dir}/scripts/prepare_peripheral_oled.py" "${absolute}" "${generated}"
            RESULT_VARIABLE result)
          if(NOT result EQUAL 0)
            message(FATAL_ERROR "LiTom peripheral OLED patch failed")
          endif()
          get_filename_component(widget_dir "${absolute}" DIRECTORY)
          target_include_directories(${library} PRIVATE "${widget_dir}")
          list(APPEND updated_sources "${generated}")
          math(EXPR replaced "${replaced}+1")
          set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
            "${absolute}" "${litom_dir}/scripts/prepare_peripheral_oled.py")
        else()
          list(APPEND updated_sources "${source}")
        endif()
      endforeach()
      set_property(TARGET ${library} PROPERTY SOURCES "${updated_sources}")
    endforeach()
    if(NOT replaced EQUAL 1)
      message(FATAL_ERROR "LiTom expected one tom_oled peripheral status source")
    endif()
  endfunction()
  cmake_language(DEFER DIRECTORY "${CMAKE_SOURCE_DIR}" CALL litom_patch_peripheral_oled)
endif()

# Keep the pinned dependency checkout untouched. Compile a corrected copy with
# timer/reconfiguration fixes required when using Studio's temporary layer UI.
if(CONFIG_ZMK_RUNTIME_INPUT_PROCESSOR)
  set_property(GLOBAL PROPERTY LITOM_MODULE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
  function(litom_patch_runtime_processor)
    get_property(litom_dir GLOBAL PROPERTY LITOM_MODULE_DIR)
    get_target_property(app_sources app SOURCES)
    set(runtime_sources ${app_sources})
    list(FILTER runtime_sources INCLUDE REGEX "/src/pointing/input_processor_runtime\\.c$")
    list(LENGTH runtime_sources runtime_count)
    if(NOT runtime_count EQUAL 1)
      message(FATAL_ERROR "LiTom expected one runtime input processor source")
    endif()
    list(GET runtime_sources 0 original)
    set(generated "${CMAKE_BINARY_DIR}/litom/input_processor_runtime.c")
    execute_process(
      COMMAND ${PYTHON_EXECUTABLE} "${litom_dir}/scripts/prepare_runtime_processor.py"
              "${original}" "${generated}"
      RESULT_VARIABLE result)
    if(NOT result EQUAL 0)
      message(FATAL_ERROR "LiTom runtime processor compatibility patch failed")
    endif()
    list(REMOVE_ITEM app_sources "${original}")
    set_property(TARGET app PROPERTY SOURCES "${app_sources}")
    target_sources(app PRIVATE "${generated}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
      "${original}" "${litom_dir}/scripts/prepare_runtime_processor.py")
  endfunction()
  cmake_language(DEFER DIRECTORY "${CMAKE_SOURCE_DIR}" CALL litom_patch_runtime_processor)
endif()
